/* ============================================================ MPL — mpl-backtest.js BTM (Backtest Modular de Radar) e BTJ (Backtest de Jogos) Regra absoluta: para avaliar concurso T, usar apenas dados < T. v1.1 — BTM aceita "passo" de amostragem (usado pelo Otimizador de K na aba Radar, para permitir comparar vários valores de K rapidamente). ============================================================ */ /* BTM — avalia cobertura do radar reduzido concurso a concurso. passo permite amostrar concursos (ex.: passo=5 testa 1 a cada 5) para viabilizar comparações rápidas entre múltiplos valores de K. */ function executarBTM(base, modalidade, config, minHistorico = 40, passo = 1) { const resultados = []; for (let i = minHistorico; i < base.length; i += passo) { const historico = base.slice(0, i); const alvo = base[i]; const { ranking } = gerarRankingDezenas(historico, modalidade, config); const rr = gerarRR(ranking, config.K, config.su.fixas, config.su.proibidas); const dezenasRR = rr.map(d => d.dezena); const cobertura = alvo.dezenas.filter(d => dezenasRR.includes(d)).length; resultados.push({ concurso: alvo.numero, cobertura, total: alvo.dezenas.length, coberturaPct: alvo.dezenas.length ? cobertura / alvo.dezenas.length : 0, tamanhoRR: dezenasRR.length }); } const coberturaPcts = resultados.map(r => r.coberturaPct); const rc = calcularRC(config.K, modalidade.dezenasSorteadas, modalidade.universo); return { resultados, coberturaMedia: media(coberturaPcts), coberturaTotal: resultados.filter(r => r.cobertura === r.total).length / Math.max(1, resultados.length), coberturaS1: resultados.filter(r => r.cobertura >= r.total - 1).length / Math.max(1, resultados.length), rc, ier: calcularIER(media(coberturaPcts), rc), amostras: resultados.length }; } /* Otimizador de K — roda o BTM para uma lista de valores de K e compara cobertura x redução combinatória, ajudando a escolher o MENOR radar com desempenho histórico aceitável (essência filosófica do MPL). */ function executarOtimizadorK(base, modalidade, config, valoresK, passo = 5, minHistorico = 40) { const resultados = [];
