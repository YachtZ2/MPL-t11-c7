/* ============================================================ MPL — mpl-scoring.js SIC, SF, SA, SC, SRT (Teia), SEI, SU, STF, CAM/FAM/RAP, RQR/RC/IER Fórmulas canônicas — Volume 0 e Volume 1 ============================================================ */ /* ---------------- SIC — Score Inicial de Camadas ---------------- */ function calcularSIC(dezena, base, camadas) { let sicBase = 0; const freqPorCamada = {}; let menorCamadaPresente = null; camadas.forEach(c => { const janela = base.slice(Math.max(0, base.length - c.janela)); const presente = janela.some(x => x.dezenas.includes(dezena)); const freq = janela.filter(x => x.dezenas.includes(dezena)).length; freqPorCamada[c.id] = freq; if (presente) { sicBase += c.peso; if (menorCamadaPresente === null) menorCamadaPresente = c.id; } }); const pesosBFC = { C1: 2.00, C2: 1.50, C3: 1.20, C4: 0.80, C5: 0.50 }; let bfc = 0; camadas.forEach(c => { bfc += (pesosBFC[c.id] || 0) * (freqPorCamada[c.id] || 0); }); const bfcLimitado = Math.min(bfc, 20); const escalaPAR = { C1: 0, C2: -3, C3: -7, C4: -12, C5: -18, NENHUMA: -25 }; const par = escalaPAR[menorCamadaPresente] ?? if (saValor !== null) { const cca = 0.70 + 0.30 * (saValor / 100); scBase = scBase * cca; } return { sc: limitar(scBase, 0, 100), spcic, rct, erc, tcr, crc, cicloMedio, cv }; } /* ---------------- SRT — Score de Relevância na Teia ---------------- */ function construirTeia(base) { const contagemIndividual = {};
